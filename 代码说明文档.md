### 1. `MiniMindConfig` 配置类 (第 10-44 行)

```python
class MiniMindConfig(PretrainedConfig):
    model_type = "minimind"
    
    # 核心参数
    hidden_size = 768           # 隐藏层维度
    num_hidden_layers = 8       # Transformer 层数
    use_moe = False             # 是否使用 MoE 架构
    
    # 注意力参数
    num_attention_heads = 8     # Q 头数量
    num_key_value_heads = 4     # KV 头数量 (GQA: 8:4=2:1)
    head_dim = hidden_size // num_attention_heads
    
    # 训练参数
    vocab_size = 6400           # 词表大小
    max_position_embeddings = 32768  # 最大上下文
    rms_norm_eps = 1e-6         # 归一化 epsilon
    rope_theta = 1e6            # RoPE 基数
    
    # MoE 参数
    num_experts = 4             # 专家数量
    num_experts_per_tok = 1     # 每个 token 激活的专家数
```

**设计要点**：
- **GQA (Grouped Query Attention)**: 8 个 Q 头、4 个 KV 头，KV 缓存减少一半
- **大词表**: 6400 词表比 LLaMA 的 32000 小，更适合中文和小模型
- **长上下文**: 32K 通过 YaRN 外推实现

### 2. `RMSNorm` 归一化层 (第 49-59 行)

```python
class RMSNorm(torch.nn.Module):
    def __init__(self, dim: int, eps: float = 1e-5):
        super().__init__()
        self.eps = eps
        self.weight = nn.Parameter(torch.ones(dim))

    def norm(self, x):
        # 计算均方根：x / sqrt(mean(x²) + eps)
        return x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.eps)

    def forward(self, x):
        return (self.weight * self.norm(x.float())).type_as(x)
```

**对比 LayerNorm**:
| | LayerNorm | RMSNorm |
|---|---|---|
| 计算 | 减均值 + 除标准差 | 只除 RMS |
| 速度 | 慢 | 快 15-20% |
| 效果 | 相同 | 相同 |

**使用位置**: 每个 Attention 和 MLP 之前

### 3. RoPE 旋转位置编码 (第 61-88 行)

```python
def precompute_freqs_cis(dim, end, rope_base, rope_scaling):
    # 计算频率：1 / (base ^ (2i/d))
    freqs = 1.0 / (rope_base ** (torch.arange(0, dim, 2)[: (dim // 2)].float() / dim))
    
    # 如果启用 YaRN（长文本外推）
    if rope_scaling is not None:
        factor = rope_scaling.get("factor", 16)
        # 根据序列长度调整频率
        freqs = freqs * (1 - ramp + ramp / factor)
    
    # 生成旋转矩阵
    t = torch.arange(end, device=freqs.device)
    freqs = torch.outer(t, freqs).float()
    freqs_cos = torch.cat([torch.cos(freqs), torch.cos(freqs)], dim=-1)
    freqs_sin = torch.cat([torch.sin(freqs), torch.sin(freqs)], dim=-1)
    return freqs_cos, freqs_sin

def apply_rotary_pos_emb(q, k, cos, sin, unsqueeze_dim=1):
    def rotate_half(x): 
        # 旋转向量：[a,b,c,d] → [-c,-d,a,b]
        return torch.cat((-x[..., x.shape[-1] // 2:], x[..., : x.shape[-1] // 2]), dim=-1)
    
    # 应用旋转：q' = q*cos + rotate(q)*sin
    q_embed = ((q * cos) + (rotate_half(q) * sin)).to(q.dtype)
    k_embed = ((k * cos) + (rotate_half(k) * sin)).to(k.dtype)
    return q_embed, k_embed
```

**RoPE 原理**:
- 传统位置编码：`h' = h + pos_emb`
- RoPE 编码：`Q = W_q * h * R(pos)`（旋转向量）
- 好处：相对位置敏感，外推性更好

### 4. `Attention` 注意力机制 (第 90-132 行)

```python
class Attention(nn.Module):
    def __init__(self, config):
        super().__init__()
        # 1. 线性投影层
        self.q_proj = nn.Linear(config.hidden_size, num_heads * head_dim, bias=False)
        self.k_proj = nn.Linear(config.hidden_size, num_kv_heads * head_dim, bias=False)
        self.v_proj = nn.Linear(config.hidden_size, num_kv_heads * head_dim, bias=False)
        self.o_proj = nn.Linear(num_heads * head_dim, config.hidden_size, bias=False)
        
        # 2. 归一化层（head 级别）
        self.q_norm = RMSNorm(self.head_dim)
        self.k_norm = RMSNorm(self.head_dim)
        
        # 3. Dropout
        self.attn_dropout = nn.Dropout(config.dropout)
        self.resid_dropout = nn.Dropout(config.dropout)

    def forward(self, x, position_embeddings, past_key_value=None, ...):
        # ===== 1. QKV 投影 =====
        xq, xk, xv = self.q_proj(x), self.k_proj(x), self.v_proj(x)
        xq = xq.view(bsz, seq_len, self.n_local_heads, self.head_dim)
        xk = xk.view(bsz, seq_len, self.n_local_kv_heads, self.head_dim)
        xv = xv.view(bsz, seq_len, self.n_local_kv_heads, self.head_dim)
        
        # ===== 2. 归一化 =====
        xq, xk = self.q_norm(xq), self.k_norm(xk)
        
        # ===== 3. 应用 RoPE =====
        cos, sin = position_embeddings
        xq, xk = apply_rotary_pos_emb(xq, xk, cos, sin)
        
        # ===== 4. KV Cache (推理时) =====
        if past_key_value is not None:
            xk = torch.cat([past_key_value[0], xk], dim=1)
            xv = torch.cat([past_key_value[1], xv], dim=1)
        present = (xk, xv) if use_cache else None
        
        # ===== 5. 重复 KV 头 (GQA) =====
        xq = xq.transpose(1, 2)
        xk = repeat_kv(xk, self.n_rep).transpose(1, 2)  # 4 头→8 头
        xv = repeat_kv(xv, self.n_rep).transpose(1, 2)
        
        # ===== 6. 注意力计算 =====
        if self.flash and seq_len > 1 and past_key_value is None:
            # Flash Attention (训练时加速)
            output = F.scaled_dot_product_attention(xq, xk, xv, is_causal=True)
        else:
            # 标准 Attention
            scores = (xq @ xk.transpose(-2, -1)) / math.sqrt(self.head_dim)
            scores[:, :, :, -seq_len:] += torch.full((seq_len, seq_len), float("-inf"), device=scores.device).triu(1)
            output = F.softmax(scores.float(), dim=-1) @ xv
        
        # ===== 7. 输出投影 =====
        output = output.transpose(1, 2).reshape(bsz, seq_len, -1)
        output = self.resid_dropout(self.o_proj(output))
        return output, present
```

**GQA 图解**:
```
Q 头： [Q1, Q2, Q3, Q4, Q5, Q6, Q7, Q8]
KV 头：[K1, K2, K3, K4]
             ↓ ↓ ↓ ↓ ↓ ↓ ↓ ↓
K 分配：[K1, K1, K2, K2, K3, K3, K4, K4]  # 2:1 复制
```

### 5. `FeedForward` 前馈网络 (第 134-144 行)

```python
class FeedForward(nn.Module):
    def __init__(self, config, intermediate_size=None):
        super().__init__()
        intermediate_size = intermediate_size or config.intermediate_size
        # 三线性结构（LLaMA 风格）
        self.gate_proj = nn.Linear(config.hidden_size, intermediate_size, bias=False)
        self.up_proj = nn.Linear(config.hidden_size, intermediate_size, bias=False)
        self.down_proj = nn.Linear(intermediate_size, config.hidden_size, bias=False)
        self.act_fn = ACT2FN[config.hidden_act]  # SiLU 激活

    def forward(self, x):
        # SwiGLU: down(gate(x) * up(x))
        return self.down_proj(self.act_fn(self.gate_proj(x)) * self.up_proj(x))
```

**与传统的区别**:
- 传统：`W2 * ReLU(W1 * x)`
- LLaMA/SwiGLU: `W_down * SiLU(W_gate * x) * W_up * x`
- 效果更好，参数更多

### 6. `MOEFeedForward` 混合专家 (第 146-174 行)

```python
class MOEFeedForward(nn.Module):
    def __init__(self, config):
        super().__init__()
        # 1. 路由网络
        self.gate = nn.Linear(config.hidden_size, config.num_experts, bias=False)
        # 2. 专家列表
        self.experts = nn.ModuleList([
            FeedForward(config, config.moe_intermediate_size) 
            for _ in range(config.num_experts)
        ])

    def forward(self, x):
        batch_size, seq_len, hidden_dim = x.shape
        x_flat = x.view(-1, hidden_dim)
        
        # ===== 1. 路由计算 =====
        scores = F.softmax(self.gate(x_flat), dim=-1)
        topk_weight, topk_idx = torch.topk(scores, k=config.num_experts_per_tok, dim=-1)
        
        if config.norm_topk_prob:
            topk_weight = topk_weight / (topk_weight.sum(dim=-1, keepdim=True) + 1e-20)
        
        # ===== 2. 专家处理 =====
        y = torch.zeros_like(x_flat)
        for i, expert in enumerate(self.experts):
            mask = (topk_idx == i)
            if mask.any():
                token_idx = mask.any(dim=-1).nonzero().flatten()
                weight = topk_weight[mask].view(-1, 1)
                y.index_add_(0, token_idx, expert(x_flat[token_idx]) * weight)
        
        # ===== 3. 辅助损失（负载均衡） =====
        load = F.one_hot(topk_idx, config.num_experts).float().mean(0)
        self.aux_loss = (load * scores.mean(0)).sum() * config.num_experts * config.router_aux_loss_coef
        
        return y.view(batch_size, seq_len, hidden_dim)
```

**MoE 工作原理**:
```
Token:  [T1, T2, T3, T4, T5, T6, T7, T8]
专家选择：[E1, E1, E2, E3, E2, E4, E1, E4]

E1 处理 [T1, T2, T7]
E2 处理 [T3, T5]
E3 处理 [T4]
E4 处理 [T6, T8]
```

**优势**: 总参数 4x，但每次只激活 1x

### 7. `MiniMindBlock` Transformer 块 (第 176-192 行)

```python
class MiniMindBlock(nn.Module):
    def __init__(self, layer_id, config):
        super().__init__()
        self.self_attn = Attention(config)
        self.input_layernorm = RMSNorm(config.hidden_size)
        self.post_attention_layernorm = RMSNorm(config.hidden_size)
        self.mlp = FeedForward(config) if not config.use_moe else MOEFeedForward(config)

    def forward(self, hidden_states, position_embeddings, past_key_value=None, ...):
        residual = hidden_states  # 保存残差
        
        # 1. Attention + Norm + 残差
        hidden_states, present = self.self_attn(
            self.input_layernorm(hidden_states),
            position_embeddings, past_key_value, use_cache, attention_mask
        )
        hidden_states += residual
        
        # 2. MLP + Norm + 残差
        residual = hidden_states
        hidden_states = hidden_states + self.mlp(self.post_attention_layernorm(hidden_states))
        
        return hidden_states, present
```

**Pre-Norm 架构**:
```
x → Norm → Attention →+ → Norm → MLP →+
```

### 8. `MiniMindModel` 主干 (第 194-226 行)

```python
class MiniMindModel(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.embed_tokens = nn.Embedding(config.vocab_size, config.hidden_size)
        self.dropout = nn.Dropout(config.dropout)
        self.layers = nn.ModuleList([MiniMindBlock(l, config) for l in range(config.num_hidden_layers)])
        self.norm = RMSNorm(config.hidden_size)
        
        # 预计算 RoPE 频率
        freqs_cos, freqs_sin = precompute_freqs_cis(...)
        self.register_buffer("freqs_cos", freqs_cos, persistent=False)
        self.register_buffer("freqs_sin", freqs_sin, persistent=False)

    def forward(self, input_ids, attention_mask=None, past_key_values=None, ...):
        # 1. 词嵌入
        hidden_states = self.dropout(self.embed_tokens(input_ids))
        
        # 2. 获取位置编码
        start_pos = past_key_values[0][0].shape[1] if past_key_values else 0
        position_embeddings = (
            self.freqs_cos[start_pos:start_pos + seq_length],
            self.freqs_sin[start_pos:start_pos + seq_length]
        )
        
        # 3. 逐层处理
        presents = []
        for layer, past_key_value in zip(self.layers, past_key_values or [None]*len(self.layers)):
            hidden_states, present = layer(hidden_states, position_embeddings, past_key_value, use_cache)
            presents.append(present)
        
        # 4. 最终归一化
        hidden_states = self.norm(hidden_states)
        
        # 5. MoE 辅助损失
        aux_loss = sum([l.mlp.aux_loss for l in self.layers if isinstance(l.mlp, MOEFeedForward)])
        
        return hidden_states, presents, aux_loss
```

### 9. `MiniMindForCausalLM` 完整模型 (第 228-279 行)

```python
class MiniMindForCausalLM(PreTrainedModel, GenerationMixin):
    config_class = MiniMindConfig
    
    def __init__(self, config):
        self.config = config or MiniMindConfig()
        super().__init__(self.config)
        self.model = MiniMindModel(self.config)
        self.lm_head = nn.Linear(self.config.hidden_size, self.config.vocab_size, bias=False)
        self.model.embed_tokens.weight = self.lm_head.weight  # 权重共享
    
    def forward(self, input_ids, labels=None, **kwargs):
        hidden_states, past_key_values, aux_loss = self.model(input_ids, **kwargs)
        logits = self.lm_head(hidden_states)
        
        loss = None
        if labels is not None:
            loss = F.cross_entropy(logits.view(-1, logits.size(-1)), labels.view(-1), ignore_index=-100)
        
        return MoeCausalLMOutputWithPast(loss=loss, aux_loss=aux_loss, logits=logits, past_key_values=past_key_values)
    
    @torch.inference_mode()
    def generate(self, inputs, max_new_tokens=8192, temperature=0.85, top_p=0.95, top_k=50, ...):
        for _ in range(max_new_tokens):
            # 1. 前向传播（只用最后一个 token，KV cache 加速）
            past_len = past_key_values[0][0].shape[1] if past_key_values else 0
            outputs = self.forward(input_ids[:, past_len:], past_key_values=past_key_values, use_cache=True)
            
            # 2. 采样
            logits = outputs.logits[:, -1, :] / temperature
            if top_k > 0:
                logits[logits < torch.topk(logits, top_k)[0][..., -1, None]] = -float('inf')
            next_token = torch.multinomial(torch.softmax(logits, dim=-1), 1)
            
            # 3. 拼接
            input_ids = torch.cat([input_ids, next_token], dim=-1)
            finished |= next_token.squeeze(-1).eq(eos_token_id)
            if finished.all():
                break
        
        return input_ids
```

## 四、训练流程 (`train_pretrain.py`)

### 训练循环 (第 23-80 行)

```python
def train_epoch(epoch, loader, iters, ...):
    for step, (input_ids, labels) in enumerate(loader):
        # 1. 前向传播 (混合精度)
        with autocast_ctx:
            res = model(input_ids, labels=labels)
            loss = res.loss / accumulation_steps
        
        # 2. 反向传播
        scaler.scale(loss).backward()
        
        # 3. 梯度更新
        if step % accumulation_steps == 0:
            scaler.unscale_(optimizer)
            clip_grad_norm_(model.parameters(), grad_clip)
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad()
```

### 关键组件

| 组件 | 作用 |
|------|------|
| `GradScaler` | 混合精度训练防止梯度下溢 |
| `clip_grad_norm_` | 梯度裁剪防止爆炸 |
| `accumulation_steps` | 梯度累积等效增大 batch |

## 六、核心技术总结

| 技术 | 位置 | 作用 |
|------|------|------|
| **GQA** | Attention | 减少 KV 缓存，加速推理 |
| **RoPE** | 位置编码 | 相对位置敏感，支持外推 |
| **SwiGLU** | FeedForward | 更好的激活函数 |
| **RMSNorm** | 归一化 | 更快更稳定 |
| **Pre-Norm** | Block | 稳定深度网络训练 |
| **MoE** | FeedForward | 稀疏激活，参数效率更高 |
| **Flash Attention** | 注意力 | 加速训练 |
| **YaRN** | 位置编码 | 长文本外推 |
